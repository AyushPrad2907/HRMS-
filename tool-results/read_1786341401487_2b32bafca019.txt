   350→function shortDate(s: string): string {
   351→  const d = new Date(s);
   352→  if (isNaN(d.getTime())) return s;
   353→  return new Intl.DateTimeFormat("en-GB", { day: "2-digit", month: "short" }).format(d);
   354→}
   355→
   356→function toDateInputValue(iso: string | null | undefined): string {
   357→  if (!iso) return "";
   358→  const d = new Date(iso);
   359→  if (isNaN(d.getTime())) return "";
   360→  const y = d.getFullYear();
   361→  const m = String(d.getMonth() + 1).padStart(2, "0");
   362→  const day = String(d.getDate()).padStart(2, "0");
   363→  return `${y}-${m}-${day}`;
   364→}

   366→function initials(name: string): string {
   367→  return name
   368→    .split(" ")
   369→    .map((p) => p[0])
   370→    .filter(Boolean)
   371→    .slice(0, 2)
   372→    .join("")
   373→    .toUpperCase();
   374→}

   376→/**
   377→ * Tiny data-fetching hook — `null` URL skips fetching. Stale data stays
   378→ * visible while `loading` is true on subsequent refetches (the skeleton only
   379→ * shows when we have no data yet).
   380→ */
   381→function useFetch<T>(url: string | null) {
   382→  const [data, setData] = React.useState<T | null>(null);
   383→  const [loading, setLoading] = React.useState<boolean>(url !== null);
   384→  const [error, setError] = React.useState<Error | null>(null);
   385→
   386→  const refetch = React.useCallback(async () => {
   387→    if (!url) {
   388→      setData(null);
   389→      setLoading(false);
   390→      return;
   391→    }
   392→    setLoading(true);
   393→    setError(null);
   394→    try {
   395→      const r = await fetch(url, { headers: { Accept: "application/json" } });
   396→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   397→      const j = (await r.json()) as T;
   398→      setData(j);
   399→    } catch (e) {
   400→      setError(e as Error);
   401→    } finally {
   402→      setLoading(false);
   403→    }
   404→  }, [url]);
   405→
   406→  React.useEffect(() => {
   407→    void refetch();
   408→  }, [refetch]);
   409→
   410→  return { data, loading, error, refetch };
   411→}

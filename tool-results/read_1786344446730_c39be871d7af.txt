   356→   356→function shortDate(s: string): string {

   358→   358→  if (isNaN(d.getTime())) return s;
   359→   359→  return new Intl.DateTimeFormat("en-GB", { day: "2-digit", month: "short" }).format(d);
   360→   360→}

   362→   362→function toDateInputValue(iso: string | null | undefined): string {
   363→   363→  if (!iso) return "";
   364→   364→  const d = new Date(iso);
   365→   365→  if (isNaN(d.getTime())) return "";
   366→   366→  const y = d.getFullYear();
   367→   367→  const m = String(d.getMonth() + 1).padStart(2, "0");
   368→   368→  const day = String(d.getDate()).padStart(2, "0");
   369→   369→  return `${y}-${m}-${day}`;
   370→   370→}

   382→   382→/**
   383→   383→ * Tiny data-fetching hook — `null` URL skips fetching. Stale data stays
   384→   384→ * visible while `loading` is true on subsequent refetches (the skeleton only
   385→   385→ * shows when we have no data yet).
   386→   386→ */
   387→   387→function useFetch<T>(url: string | null) {
   388→   388→  const [data, setData] = React.useState<T | null>(null);
   389→   389→  const [loading, setLoading] = React.useState<boolean>(url !== null);
   390→   390→  const [error, setError] = React.useState<Error | null>(null);
   391→   391→
   392→   392→  const refetch = React.useCallback(async () => {
   393→   393→    if (!url) {
   394→   394→      setData(null);
   395→   395→      setLoading(false);
   396→   396→      return;
   397→   397→    }
   398→   398→    setLoading(true);
   399→   399→    setError(null);
   400→   400→    try {
   401→   401→      const r = await fetch(url, { headers: { Accept: "application/json" } });
   402→   402→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   403→   403→      const j = (await r.json()) as T;
   404→   404→      setData(j);
   405→   405→    } catch (e) {
   406→   406→      setError(e as Error);
   407→   407→    } finally {
   408→   408→      setLoading(false);
   409→   409→    }
   410→   410→  }, [url]);
   411→   411→
   412→   412→  React.useEffect(() => {
   413→   413→    void refetch();
   414→   414→  }, [refetch]);
   415→   415→
   416→   416→  return { data, loading, error, refetch };
   417→   417→}

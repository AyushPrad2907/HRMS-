   357→function shortDate(s: string): string {
   358→  const d = new Date(s);
   359→  if (isNaN(d.getTime())) return s;
   360→  return new Intl.DateTimeFormat("en-GB", { day: "2-digit", month: "short" }).format(d);
   361→}

   363→function toDateInputValue(iso: string | null | undefined): string {
   364→  if (!iso) return "";
   365→  const d = new Date(iso);
   366→  if (isNaN(d.getTime())) return "";
   367→  const y = d.getFullYear();
   368→  const m = String(d.getMonth() + 1).padStart(2, "0");
   369→  const day = String(d.getDate()).padStart(2, "0");
   370→  return `${y}-${m}-${day}`;
   371→}

   380→    .toUpperCase();
   381→}

   383→/**
   384→ * Tiny data-fetching hook — `null` URL skips fetching. Stale data stays
   385→ * visible while `loading` is true on subsequent refetches (the skeleton only
   386→ * shows when we have no data yet).
   387→ */
   388→function useFetch<T>(url: string | null) {
   389→  const [data, setData] = React.useState<T | null>(null);
   390→  const [loading, setLoading] = React.useState<boolean>(url !== null);
   391→  const [error, setError] = React.useState<Error | null>(null);
   392→
   393→  const refetch = React.useCallback(async () => {
   394→    if (!url) {
   395→      setData(null);
   396→      setLoading(false);
   397→      return;
   398→    }
   399→    setLoading(true);
   400→    setError(null);
   401→    try {
   402→      const j = await apiFetch<T>(url, { headers: { Accept: "application/json" } });
   403→      setData(j);
   404→    } catch (e) {
   405→      setError(e instanceof ApiError ? new Error(`HTTP ${e.status}`) : (e as Error));
   406→    } finally {
   407→      setLoading(false);
   408→    }
   409→  }, [url]);
   410→
   411→  React.useEffect(() => {
   412→    void refetch();
   413→  }, [refetch]);
   414→
   415→  return { data, loading, error, refetch };
   416→}

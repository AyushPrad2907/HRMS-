   356→function shortDate(s: string): string {
   357→  const d = new Date(s);
   358→  if (isNaN(d.getTime())) return s;
   359→  return new Intl.DateTimeFormat("en-GB", { day: "2-digit", month: "short" }).format(d);
   360→}

   362→function toDateInputValue(iso: string | null | undefined): string {
   363→  if (!iso) return "";
   364→  const d = new Date(iso);
   365→  if (isNaN(d.getTime())) return "";
   366→  const y = d.getFullYear();
   367→  const m = String(d.getMonth() + 1).padStart(2, "0");
   368→  const day = String(d.getDate()).padStart(2, "0");
   369→  return `${y}-${m}-${day}`;
   370→}

   372→function initials(name: string): string {
   373→  return name
   374→    .split(" ")
   375→    .map((p) => p[0])

   377→    .slice(0, 2)
   378→    .join("")
   379→    .toUpperCase();
   380→}

   382→/**
   383→ * Tiny data-fetching hook — `null` URL skips fetching. Stale data stays
   384→ * visible while `loading` is true on subsequent refetches (the skeleton only
   385→ * shows when we have no data yet).
   386→ */
   387→function useFetch<T>(url: string | null) {
   388→  const [data, setData] = React.useState<T | null>(null);
   389→  const [loading, setLoading] = React.useState<boolean>(url !== null);
   390→  const [error, setError] = React.useState<Error | null>(null);
   391→
   392→  const refetch = React.useCallback(async () => {
   393→    if (!url) {
   394→      setData(null);
   395→      setLoading(false);
   396→      return;
   397→    }
   398→    setLoading(true);
   399→    setError(null);
   400→    try {
   401→      const r = await fetch(url, { headers: { Accept: "application/json" } });
   402→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   403→      const j = (await r.json()) as T;
   404→      setData(j);
   405→    } catch (e) {
   406→      setError(e as Error);
   407→    } finally {
   408→      setLoading(false);
   409→    }
   410→  }, [url]);
   411→
   412→  React.useEffect(() => {
   413→    void refetch();
   414→  }, [refetch]);
   415→
   416→  return { data, loading, error, refetch };
   417→}

   167→function todayISODate(): string {
   168→  const d = new Date();
   169→  const y = d.getFullYear();
   170→  const m = String(d.getMonth() + 1).padStart(2, "0");
   171→  const day = String(d.getDate()).padStart(2, "0");
   172→  return `${y}-${m}-${day}`;
   173→}

   181→function parseLeaveBalances(items: SettingItem[] | undefined): LeaveBalances {
   182→  if (!items) return EMPTY_BALANCES;
   183→  const row = items.find((s) => s.key === "leave_balances");
   184→  if (!row) return EMPTY_BALANCES;
   185→  const v = row.value;
   186→  if (v && typeof v === "object" && !Array.isArray(v)) {
   187→    const obj = v as Partial<Record<keyof LeaveBalances, unknown>>;
   188→    const num = (x: unknown): number =>
   189→      typeof x === "number" && isFinite(x) ? x : typeof x === "string" ? Number(x) || 0 : 0;
   190→    return {
   191→      casual: num(obj.casual),
   192→      sick: num(obj.sick),
   193→      earned: num(obj.earned),
   194→      unpaid: num(obj.unpaid),
   195→    };
   196→  }
   197→  return EMPTY_BALANCES;
   198→}

   200→/**
   201→ * Tiny data-fetching hook — `null` URL skips fetching. Stale data stays
   202→ * visible while `loading` is true on subsequent refetches (the skeleton only
   203→ * shows when we have no data yet).
   204→ */
   205→function useFetch<T>(url: string | null) {
   206→  const [data, setData] = React.useState<T | null>(null);
   207→  const [loading, setLoading] = React.useState<boolean>(url !== null);
   208→  const [error, setError] = React.useState<Error | null>(null);
   209→
   210→  const refetch = React.useCallback(async () => {
   211→    if (!url) {
   212→      setData(null);
   213→      setLoading(false);
   214→      return;
   215→    }
   216→    setLoading(true);
   217→    setError(null);
   218→    try {
   219→      const r = await fetch(url, { headers: { Accept: "application/json" } });
   220→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   221→      const j = (await r.json()) as T;
   222→      setData(j);
   223→    } catch (e) {
   224→      setError(e as Error);
   225→    } finally {
   226→      setLoading(false);
   227→    }
   228→  }, [url]);
   229→
   230→  React.useEffect(() => {
   231→    void refetch();
   232→  }, [refetch]);
   233→
   234→  return { data, loading, error, refetch };
   235→}

   286→// ---------------------------------------------------------------------------

   290→function AttendanceActionButtons({
   291→  attendanceToday,
   292→  onDone,
   293→}: {
   294→  attendanceToday: AttendanceToday | null;
   295→  onDone: () => void;
   296→}) {
   297→  const [busy, setBusy] = React.useState(false);
   298→  const checkedIn = !!attendanceToday?.checkIn;
   299→  const checkedOut = !!attendanceToday?.checkOut;
   300→
   301→  async function act(action: "check_in" | "check_out") {
   302→    setBusy(true);
   303→    try {
   304→      const r = await fetch("/api/attendance", {
   305→        method: "POST",
   306→        headers: { "Content-Type": "application/json" },
   307→        body: JSON.stringify({ action }),
   308→      });
   309→      if (!r.ok) {
   310→        const j = (await r.json().catch(() => null)) as { error?: string } | null;
   311→        throw new Error(j?.error ?? `HTTP ${r.status}`);
   312→      }
   313→      toast.success(action === "check_in" ? "Checked in" : "Checked out");
   314→      onDone();
   315→    } catch (e) {
   316→      toast.error((e as Error).message);
   317→    } finally {
   318→      setBusy(false);
   319→    }
   320→  }
   321→
   322→  if (!attendanceToday || !checkedIn) {
   323→    return (
   324→      <Button onClick={() => act("check_in")} disabled={busy}>
   325→        <LogIn className="size-4" />
   326→        {busy ? "Checking in…" : "Check In"}
   327→      </Button>
   328→    );
   329→  }
   330→  if (checkedIn && !checkedOut) {
   331→    return (
   332→      <Button onClick={() => act("check_out")} disabled={busy}>
   333→        <LogOut className="size-4" />
   334→        {busy ? "Checking out…" : "Check Out"}
   335→      </Button>
   336→    );
   337→  }
   338→  return (
   339→    <Badge variant="outline" className="gap-1.5">
   340→      <Check className="size-3.5 text-emerald-600 dark:text-emerald-400" />
   341→      Day complete
   342→    </Badge>
   343→  );
   344→}

   483→type ClassFormRow = {
   484→  batch: string;
   485→  subject: string;
   486→  topicsCovered: string;
   487→  studentsAttended: string;
   488→  assignmentsChecked: boolean;
   489→};
   490→
   491→const EMPTY_CLASS_ROW: ClassFormRow = {
   492→  batch: "",
   493→  subject: "",
   494→  topicsCovered: "",
   495→  studentsAttended: "",
   496→  assignmentsChecked: false,
   497→};
   498→
   499→function DailyReportSection({
   500→  onSubmitted,
   501→}: {
   502→  onSubmitted: () => void;
   503→}) {
   504→  const [reportDate, setReportDate] = React.useState<string>(todayISODate());
   505→  const [notes, setNotes] = React.useState<string>("");
   506→  const [classes, setClasses] = React.useState<ClassFormRow[]>([
   507→    { ...EMPTY_CLASS_ROW },
   508→  ]);
   509→  const [submitting, setSubmitting] = React.useState(false);
   510→
   511→  function updateRow(i: number, patch: Partial<ClassFormRow>) {
   512→    setClasses((prev) =>
   513→      prev.map((row, idx) => (idx === i ? { ...row, ...patch } : row)),
   514→    );
   515→  }
   516→  function addRow() {
   517→    setClasses((prev) => [...prev, { ...EMPTY_CLASS_ROW }]);
   518→  }
   519→  function removeRow(i: number) {
   520→    setClasses((prev) =>
   521→      prev.length === 1 ? prev : prev.filter((_, idx) => idx !== i),
   522→    );
   523→  }
   524→
   525→  function validate(): string | null {
   526→    if (!reportDate) return "Pick a report date.";
   527→    for (let i = 0; i < classes.length; i++) {
   528→      const r = classes[i];
   529→      if (!r.batch.trim()) return `Class #${i + 1}: batch is required.`;
   530→      if (!r.subject.trim()) return `Class #${i + 1}: subject is required.`;
   531→      if (!r.topicsCovered.trim()) return `Class #${i + 1}: topics covered is required.`;
   532→      const n = Number(r.studentsAttended);
   533→      if (!r.studentsAttended || !isFinite(n) || n <= 0)
   534→        return `Class #${i + 1}: students attended must be a positive number.`;
   535→    }
   536→    return null;
   537→  }
   538→
   539→  async function submit() {
   540→    const err = validate();
   541→    if (err) {
   542→      toast.error(err);
   543→      return;
   544→    }
   545→    setSubmitting(true);
   546→    try {
   547→      const payload = {
   548→        reportDate,
   549→        notes: notes.trim() || null,
   550→        classes: classes.map((c) => ({
   551→          batch: c.batch.trim(),
   552→          subject: c.subject.trim(),
   553→          topicsCovered: c.topicsCovered.trim(),
   554→          studentsAttended: Number(c.studentsAttended),
   555→          assignmentsChecked: c.assignmentsChecked,
   556→        })),
   557→      };
   558→      const r = await fetch("/api/teacher-reports", {
   559→        method: "POST",
   560→        headers: { "Content-Type": "application/json" },
   561→        body: JSON.stringify(payload),
   562→      });
   563→      if (r.status === 409) {
   564→        const j = (await r.json().catch(() => null)) as { error?: string } | null;
   565→        toast.error(j?.error ?? "Report already submitted for this date.");
   566→        return;
   567→      }
   568→      if (!r.ok) {
   569→        const j = (await r.json().catch(() => null)) as { error?: string } | null;
   570→        throw new Error(j?.error ?? `HTTP ${r.status}`);
   571→      }
   572→      toast.success("Report submitted");
   573→      setNotes("");
   574→      setClasses([{ ...EMPTY_CLASS_ROW }]);
   575→      setReportDate(todayISODate());
   576→      onSubmitted();
   577→    } catch (e) {
   578→      toast.error((e as Error).message);
   579→    } finally {
   580→      setSubmitting(false);
   581→    }
   582→  }
   583→
   584→  return (
   585→    <div className="space-y-6">
   586→      <SectionHeader title="Daily Report" description="Log the classes you taught today" />
   587→
   588→      <Card className="px-6 py-5">
   589→        <CardContent className="px-0 space-y-6">
   590→          {/* Date + Notes */}
   591→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   592→            <div className="space-y-1.5">
   593→              <Label htmlFor="report-date">Report Date</Label>
   594→              <Input
   595→                id="report-date"
   596→                type="date"
   597→                value={reportDate}
   598→                max={todayISODate()}
   599→                onChange={(e) => setReportDate(e.target.value)}
   600→              />
   601→            </div>
   602→            <div className="space-y-1.5">
   603→              <Label htmlFor="report-notes">Notes (optional)</Label>
   604→              <Input
   605→                id="report-notes"
   606→                placeholder="Any general notes for the day"
   607→                value={notes}
   608→                onChange={(e) => setNotes(e.target.value)}
   609→              />
   610→            </div>
   611→          </div>
   612→
   613→          {/* Classes */}
   614→          <div className="space-y-3">
   615→            <div className="flex items-center justify-between">
   616→              <h3 className="text-sm font-semibold">Classes</h3>
   617→              <Button variant="outline" size="sm" onClick={addRow}>
   618→                <Plus className="size-4" />
   619→                Add Class
   620→              </Button>
   621→            </div>
   622→
   623→            <div className="space-y-4">
   624→              {classes.map((row, i) => (
